EXERCISE 1.4 — EXACT ODEs. INTEGRATING FACTORS
04\(e^{3\theta}\left(dr+3r\,d\theta\right)=0\)완전미분
\[e^{3\theta}\,dr+3re^{3\theta}\,d\theta=0\]
\[M=e^{3\theta},\qquad N=3re^{3\theta}\]
\[\frac{\partial M}{\partial\theta}=3e^{3\theta}=\frac{\partial N}{\partial r}\quad\Longrightarrow\quad\text{exact}\]
\[u=\int M\,dr=re^{3\theta}+\phi(\theta)\]
\[\frac{\partial u}{\partial\theta}=3re^{3\theta}+\phi'(\theta)=N=3re^{3\theta}\]
\[\phi'(\theta)=0\ \Longrightarrow\ \phi(\theta)=c\]
\(re^{3\theta}=c\)
check
\[d\!\left(re^{3\theta}\right)=e^{3\theta}dr+3re^{3\theta}d\theta=0\quad\checkmark\]
11\(2\cosh x\cos y\,dx=\sinh x\sin y\,dy\)적분인자 F(x)
\[2\cosh x\cos y\,dx-\sinh x\sin y\,dy=0\]
\[P=2\cosh x\cos y,\qquad Q=-\sinh x\sin y\]
\[P_{y}=-2\cosh x\sin y,\qquad Q_{x}=-\cosh x\sin y\ \Longrightarrow\ P_{y}\neq Q_{x}\]
integrating factor
\[R(x)=\frac{1}{Q}\left(P_{y}-Q_{x}\right)=\frac{-\cosh x\sin y}{-\sinh x\sin y}=\coth x\]
\[\frac{F'}{F}=R(x)=\coth x\ \Longrightarrow\ F=\sinh x\qquad(x\neq 0)\]
\[2\sinh x\cosh x\cos y\,dx-\sinh^{2}x\sin y\,dy=0\]
\[u=\int 2\sinh x\cosh x\cos y\,dx=\sinh^{2}x\cos y+\phi(y)\]
\[\frac{\partial u}{\partial y}=-\sinh^{2}x\sin y+\phi'(y)=-\sinh^{2}x\sin y\ \Longrightarrow\ \phi(y)=c\]
\(\sinh^{2}x\cos y=c\)
check
\[\frac{\partial}{\partial x}\!\left(\sinh^{2}x\cos y\right)=2\sinh x\cosh x\cos y\quad\checkmark\]
\[\frac{\partial}{\partial y}\!\left(\sinh^{2}x\cos y\right)=-\sinh^{2}x\sin y\quad\checkmark\]
12\(\left(2xy\,dx+dy\right)e^{x^{2}}=0,\quad y(0)=2\)완전미분 · IVP
\[2xye^{x^{2}}dx+e^{x^{2}}dy=0\]
\[M=2xye^{x^{2}},\qquad N=e^{x^{2}}\]
\[M_{y}=2xe^{x^{2}}=N_{x}\quad\Longrightarrow\quad\text{exact}\]
\[u=\int 2xye^{x^{2}}dx=ye^{x^{2}}+\phi(y)\]
\[\frac{\partial u}{\partial y}=e^{x^{2}}+\phi'(y)=N=e^{x^{2}}\ \Longrightarrow\ \phi(y)=c\]
\[ye^{x^{2}}=c\]
initial condition
\[x=0,\ y=2:\quad 2e^{0}=2\ \Longrightarrow\ c=2\]
\(y=2e^{-x^{2}}\)
check
\[y(0)=2e^{0}=2\quad\checkmark\]
\[y'=-4xe^{-x^{2}},\qquad 2xy=4xe^{-x^{2}}\ \Longrightarrow\ y'+2xy=0\quad\checkmark\]

09\(y'+y\sin x=e^{\cos x},\quad y(0)=-2.5\)선형 비제차 · IVP
\[p(x)=\sin x,\qquad r(x)=e^{\cos x}\]
\[h=\int p\,dx=\int\sin x\,dx=-\cos x\]
\[y=e^{-h}\left[\int e^{h}r\,dx+c\right]=e^{\cos x}\left[\int e^{-\cos x}e^{\cos x}dx+c\right]\]
\[=e^{\cos x}\left[\int dx+c\right]=(x+c)\,e^{\cos x}\]
initial condition
\[y(0)=(0+c)e^{\cos 0}=ce=-2.5\ \Longrightarrow\ c=-\frac{2.5}{e}\]
\(y=\left(x-\dfrac{2.5}{e}\right)e^{\cos x}\)
check
\[y'=e^{\cos x}-(x+c)\sin x\,e^{\cos x}\]
\[y'+y\sin x=e^{\cos x}\quad\checkmark\]

12\(xy'+4y=8x^{4},\quad y(1)=2\)선형 비제차 · IVP
\[y'+\frac{4}{x}\,y=8x^{3}\qquad\text{(standard form)}\]
\[p(x)=\frac{4}{x},\qquad r(x)=8x^{3}\]
\[h=\int\frac{4}{x}dx=4\ln x,\qquad e^{h}=x^{4},\qquad e^{-h}=x^{-4}\]
\[y=x^{-4}\left[\int x^{4}\cdot 8x^{3}dx+c\right]=x^{-4}\left[\int 8x^{7}dx+c\right]\]
\[=x^{-4}\left(x^{8}+c\right)=x^{4}+cx^{-4}\]
initial condition
\[y(1)=1+c=2\ \Longrightarrow\ c=1\]
\(y=x^{4}+x^{-4}\qquad(x>0)\)
check
\[y(1)=1+1=2\quad\checkmark\]
\[y'=4x^{3}-4x^{-5},\qquad xy'+4y=4x^{4}-4x^{-4}+4x^{4}+4x^{-4}=8x^{4}\quad\checkmark\]
